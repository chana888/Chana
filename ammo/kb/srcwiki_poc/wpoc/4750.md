### 漏洞描述
Integration for ActiveCampaign and Contact Form 7, WPForms, Elementor, Ninja Forms WordPress 插件存在未认证 PHP 对象注入漏洞（CWE-502: Deserialization of Untrusted Data）。cf7-active-campaign.php 中 maybe_unserialize() 直接处理用户输入，未经验证。若存在 POP 链，可导致 RCE、文件删除或数据泄露。

### POC/利用
```python
#!/usr/bin/env python3
"""CVE-2026-9691 - ActiveCampaign PHP Object Injection"""
import requests, re, sys
from bs4 import BeautifulSoup

def detect(url):
    paths = ['/wp-content/plugins/cf7-active-campaign/']
    for p in paths:
        try:
            r = requests.get(f"{url}{p}cf7-active-campaign.php", timeout=10)
            if r.status_code == 200:
                m = re.search(r'Version:\s*([0-9.]+)', r.text)
                if m:
                    v = m.group(1)
                    parts = v.split('.')
                    if len(parts) >= 3:
                        if int(parts[0]) < 1 or (int(parts[0])==1 and int(parts[1])<1) or (int(parts[0])==1 and int(parts[1])==1 and int(parts[2])<=1):
                            return True, v
        except: pass
    return False, None

if __name__ == "__main__":
    if len(sys.argv) < 2:
        print(f"Usage: {sys.argv[0]} <target>")
        sys.exit(1)
    t = sys.argv[1].rstrip('/')
    vuln, ver = detect(t)
    if vuln:
        print(f"[!] VULNERABLE v{ver} - CVE-2026-9691 PHP Object Injection")
    else:
        print("[-] Not vulnerable")
```
